\begin{frame}{Problématique}
  \begin{block}{Question centrale}
    Combien de temps met une particule quantique pour traverser une
    barrière de potentiel qu'elle n'a \emph{classiquement} pas le droit de franchir ?
  \end{block}
  \vspace{0.5em}
  \textbf{Démarche du projet :}
  \begin{enumerate}
    \item Construire un paquet d'ondes gaussien et comprendre sa dynamique libre ;
    \item Déterminer les états stationnaires de la barrière rectangulaire
          (transmission $|T|^2$, phase $\varphi_T$) ;
    \item Résoudre numériquement l'équation de Schrödinger dépendante du temps
          (schéma de Crank--Nicolson) ;
    \item Mesurer le temps de traversée et le comparer à la prédiction
          théorique (temps de groupe de Hartman) ;
    \item Étudier l'influence de la barrière ($a$, $V_0$) sur ce temps.
  \end{enumerate}
  \vspace{0.5em}
  Unités adimensionnées : $\hb = m = 1$.
\end{frame}

\begin{frame}{Motivation : de l'onde plane au paquet d'ondes}
  \begin{columns}[T]
    \begin{column}{0.44\textwidth}
      \footnotesize
      \textbf{Onde plane} — solution de Schrödinger :
      $$\Psi(x,t)=\Psi_0\,e^{i(kx-\omega t)}$$
      \vspace{0.3em}
      Mais $\int_{-\infty}^{+\infty}|\Psi|^2\,dx$ diverge :\\
      \alert{non normalisable, donc non physique.}
      \vspace{0.8em}

      \textbf{Superposition de 3 ondes planes} ($k_0$, $k_0\pm\Delta k/2$) :\\
      les interférences créent une \textbf{enveloppe localisée}.
      \vspace{0.8em}

      \textbf{Limite continue} $\to$ intégrale sur tous les $k$ :\\
      le \textbf{paquet d'ondes gaussien} (Partie 2).
    \end{column}
    \begin{column}{0.54\textwidth}
      \includegraphics[width=\linewidth]{figures/Figure_P1_superposition.png}
      \vspace{0.2em}
      {\scriptsize\color{grisdoux} Superposition de 3 ondes : l'enveloppe\\
      $A[1+\cos(\Delta k\,x/2)]$ localise la particule.}
    \end{column}
  \end{columns}
\end{frame}

% ------------------------------------------------------------
\begin{frame}{Le paquet d'ondes gaussien}
  Une particule localisée = superposition d'ondes planes :
  \[
    \psi(x,t) = \frac{1}{\sqrt{2\pi}} \int g(k)\,\eg^{i(kx - \omega(k)t)}\,dk,
    \qquad \omega(k) = \frac{\hb k^2}{2m}
  \]
  avec un poids gaussien $g(k) \propto \eg^{-a_{\rm wp}(k-k_0)^2}$ centré en $k_0$.

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Propriétés clés :}
      \begin{itemize}
        \item vitesse de groupe $v_g = \hb k_0/m$ \\
              (vitesse du centre du paquet) ;
        \item vitesse de phase $v_\varphi = v_g/2$ ;
        \item largeur spectrale $\Delta k = \dfrac{1}{2\sqrt{a_{\rm wp}}}$ \\
              $\Rightarrow$ étroit en $x$ $\Leftrightarrow$ large en $k$ ;
        \item étalement (dispersion) au cours du temps.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \includegraphics[width=\textwidth]{Figure_0_paquet_re_im.png}
      \centering {\footnotesize Parties réelle et imaginaire à $t=0$ (partie 2).}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{États stationnaires de la barrière rectangulaire}
  Solutions $\psi(x)\,\eg^{-iEt/\hb}$ pour $V(x) = V_0$ sur $[0,a]$, avec $E < V_0$ :
  \[
    \psi_{\rm I} = \eg^{ikx} + R\,\eg^{-ikx}
    \quad\Big|\quad
    \psi_{\rm II} = A\,\eg^{\kappa x} + B\,\eg^{-\kappa x}
    \quad\Big|\quad
    \psi_{\rm III} = T\,\eg^{ikx}
  \]
  \[
    k = \frac{\sqrt{2mE}}{\hb}, \qquad
    \kappa = \frac{\sqrt{2m(V_0 - E)}}{\hb}
    \quad \text{(onde évanescente sous la barrière)}
  \]
  \vspace{0.3em}
  Raccordement de $\psi$ et $\psi'$ en $x=0$ et $x=a$ $\Rightarrow$
  \begin{block}{Amplitude de transmission}
    \[
      T(k) = \frac{\eg^{-ika}}{\cosh(\kappa a)
             + i\,\dfrac{\kappa^2 - k^2}{2k\kappa}\,\sinh(\kappa a)}
      \qquad\Rightarrow\qquad
      |T|^2 = \frac{1}{1 + \dfrac{(\kappa^2+k^2)^2}{4k^2\kappa^2}\sinh^2(\kappa a)}
    \]
  \end{block}
  Régime opaque ($\kappa a \gg 1$) :
  $|T|^2 \approx \dfrac{16k^2\kappa^2}{(k^2+\kappa^2)^2}\,\eg^{-2\kappa a}$
  — décroissance \textbf{exponentielle} avec $a$.
\end{frame}

\begin{frame}[fragile]{Méthode numérique : schéma de Crank--Nicolson}
  \vspace{-0.2em}
  Équation de Schrödinger discrétisée en moyennant les instants $t_j$ et $t_{j+1}$ :
  \[
    -r\,\psi_{i-1}^{j+1} + (1 + 2r + s_i)\,\psi_i^{j+1} - r\,\psi_{i+1}^{j+1}
    = r\,\psi_{i-1}^{j} + (1 - 2r - s_i)\,\psi_i^{j} + r\,\psi_{i+1}^{j}
  \]
  \vspace{-0.6em}
  \[
    r = \frac{i\hb\,\Delta t}{4m\,\Delta x^2}, \qquad
    \tikzmarknode{eqsi}{\color{bordeaux}s_i} = \frac{i\,V(x_i)\,\Delta t}{2\hb}
  \]
  \vspace{-0.5em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \footnotesize\textbf{Pourquoi ce schéma ?}
      \begin{itemize}\footnotesize\setlength{\itemsep}{0pt}
        \item \textbf{unitaire} : la norme $\int |\psi|^2 dx$ est conservée
              à $10^{-6}$ près sur toute la simulation ;
        \item inconditionnellement stable, précis à l'ordre 2 en temps et en espace ;
        \item système tridiagonal résolu par la
              \tikzmarknode{eqthomas}{\color{bordeaux}méthode de Thomas} en $O(n_x)$
              à chaque pas de temps.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      % On repasse en footnotesize juste pour cette boîte (4 lignes au
      % lieu de 3 ailleurs) : \codetaille est redéfini localement, dans
      % un groupe ouvert par begingroup/endgroup pour ne pas gêner
      % le parsing verbatim de tcblisting.
      \begingroup
      \renewcommand{\codetaille}{\footnotesize}
      \begin{tcblisting}{codebox}
(*\tikzmarknode{codesi}{\color{codeaccent}\bfseries s\_int}*) = 1j*V_int*dt/(2*hbar)
b_mat = (1 + 2*r) + s_int
psi[1:-1] = (*\tikzmarknode{codethomas}{\color{codeaccent}\bfseries thomas}*)(
    a_mat, b_mat, c_mat, rhs)
      \end{tcblisting}
      \endgroup
      \codesource{evolution\_schrodinger() et resoudre\_tridiagonal() --- Partie 4}
    \end{column}
  \end{columns}
  \vspace{-0.05em}
  {\scriptsize\textbf{Paramètres :}
   grille $n_x{=}1500$, $n_t{=}3000$, $x{\in}[-30,30]$, $t{\in}[0,6]$ ~~
   paquet $k_0{=}5$ ($E{=}12.5$), $a_{\rm wp}{=}1$, $x_0{=}-10$ ~~
   barrière $a{=}1$, $V_0{=}15{>}E$.}
  % Une seule flèche sur cette diapo : s_i -> sa ligne de code. Pour
  % "méthode de Thomas" -> thomas(...), une flèche propre devrait traverser
  % la fin de la 3e puce ("en O(nx) à chaque pas de temps") quel que soit
  % l'angle testé, vu que les deux colonnes sont denses en texte ; on garde
  % donc seulement le lien par couleur (bordeaux/corail partagé), sans
  % tracer de flèche par-dessus le texte.
  \begin{tikzpicture}[remember picture, overlay]
    \draw[-{Triangle[length=2mm]}, thick, bordeaux, looseness=0.8]
      (eqsi.south) to[out=-90,in=90] (codesi.north);
  \end{tikzpicture}
\end{frame}

\begin{frame}[fragile]{Quel temps de traversée ? Le temps de groupe de Hartman}
  \small
  Le paquet transmis s'écrit en pondérant chaque composante par $T(k)$ :
  \vspace{-0.2em}
  \[
    \psi_{\rm trans}(x,t) = \frac{1}{\sqrt{2\pi}}\int g(k)\,T(k)\,
    \eg^{i(kx-\omega t)}\,dk
    \qquad \text{avec } T = |T|\,\eg^{i\varphi_T}
  \]
  \vspace{-0.3em}
  Phase stationnaire autour de $k_0$ $\Rightarrow$ le pic transmis est décalé de
  $\tau_g = \dfrac{1}{v_g}\dfrac{d\varphi_T}{dk}\Big|_{k_0}$ \;($< 0$ en régime tunnel) :
  \vspace{-0.2em}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \begin{block}{Temps de traversée théorique}
        \[
          \tikzmarknode{eqtaut}{\color{bordeaux}\tau_t^{\text{th}} = \tau_0 + \tau_g}
        \]
        {\scriptsize avec $\varphi_T(k) = -ka
          - \arctan\!\left[\tfrac{\kappa^2 - k^2}{2k\kappa}\tanh(\kappa a)\right]$}
      \end{block}
    \end{column}
    \begin{column}{0.40\textwidth}
      \begin{tcblisting}{codebox}
tau_g = dphi_dk / vg
tau0 = tau_libre(k0, a)
return (*\tikzmarknode{codetaut}{\color{codeaccent}\bfseries tau0 + tau\_g}*)
      \end{tcblisting}
      \codesource{tau\_tunnel() --- Partie 4}
    \end{column}
  \end{columns}
  \vspace{-0.3em}
  \textbf{Résultat} ($k_0=5$, $a=1$, $V_0=15$) :
  $\;\tau_t^{\text{num}} = 0.112 \;<\; \tau_t^{\text{th}} = 0.168 \;<\; \tau_0 = 0.200$
  — le paquet tunnel sort \emph{plus tôt} que le paquet libre :
  c'est l'\textbf{effet Hartman}.
  % codetaut ne couvre que "tau0 + tau_g", pas le mot-clé "return" qui le
  % précède sur la même ligne : on entre donc par le bas (.south) plutôt
  % que par la gauche, pour ne pas traverser "return" horizontalement.
  % Trajet en équerre (pas une courbe bézier) pour rester maître du
  % passage : on sort à droite de la formule, on descend dans la marge
  % vide sous les deux colonnes (entre le bloc, la boîte de code et la
  % ligne "Résultat"), puis on remonte pile sous "tau0 + tau_g".

  	\draw[-{Triangle[length=2mm]}, thick, bordeaux]
  	(eqtaut.east) to[out=0, in=-160, looseness=1.5] (codetaut.south);
  \end{tikzpicture}
\end{frame}

\begin{frame}{Influence de la barrière : largeur $a$}
  \begin{center}
    \includegraphics[width=0.82\textwidth]{Figure_3_influence_a.png}
  \end{center}
  \begin{itemize}
    \item $\tau_0 \propto a$ (libre) mais $\tau_t^{\text{th}}$ \textbf{sature}
          vers $0.179$ dès $\kappa a \gtrsim 3$ : le temps tunnel devient
          indépendant de l'épaisseur (Hartman, 1962).
    \item $|T|^2$ chute exponentiellement ($\sim \eg^{-2\kappa a}$),
          conformément aux états stationnaires.
    \item Au-delà de $a \approx 1.5$, la mesure par le pic décroche : le
          filtrage spectral domine — limite de cette définition du temps.
  \end{itemize}
\end{frame}

\begin{frame}{Influence de la barrière : hauteur $V_0$}
  \begin{center}
    \includegraphics[width=0.82\textwidth]{Figure_4_influence_V0.png}
  \end{center}
  \begin{itemize}
    \item Plus la barrière est haute, plus $\kappa$ est grand : $|T|^2$
          s'effondre\dots{} mais le temps de traversée \textbf{diminue} aussi
          ($0.218 \to 0.080$ pour $V_0 : 13.5 \to 25$).
    \item Résultat contre-intuitif : une barrière plus infranchissable se
          traverse (rarement) mais \emph{plus vite}.
  \end{itemize}
\end{frame}

\begin{frame}{Conclusion}
  \textbf{Ce que nous avons montré :}
  \begin{itemize}
    \item Un solveur Crank--Nicolson validé (norme conservée, $\tau_0$ à 4\,\%
          de la théorie) pour la dynamique d'un paquet gaussien face à une
          barrière.
    \item Les états stationnaires donnent $|T|^2$ et la phase $\varphi_T$ :
          la décroissance exponentielle de la transmission est vérifiée
          numériquement.
    \item Le temps de traversée mesuré confirme l'\textbf{effet Hartman} :
          $\tau_t < \tau_0$.
    \item L'effet est robuste aux caractéristiques de la \textbf{barrière} :
          $\tau_t^{\text{th}}$ sature avec $a$ et \textbf{diminue} avec $V_0$ —
          une barrière plus infranchissable se traverse (rarement) mais
          plus vite.
  \end{itemize}
  \vspace{0.5em}
  \textbf{Perspectives :} influence des caractéristiques du \emph{paquet}
  lui-même ($k_0$, $a_{\rm wp}$) sur ce temps — piste explorée puis écartée
  du périmètre final par souci de temps, mais déjà entrevue à la diapo 8
  (le filtrage spectral de la barrière) ; autres définitions du temps
  tunnel (temps de séjour, horloge de Larmor) ; double barrière ;
  comparaison aux expériences d'attophysique.
  \vspace{0.8em}
  \begin{center}
    \emph{Merci de votre attention !}
  \end{center}
\end{frame}
